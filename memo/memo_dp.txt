// ナップサックDP  N*W が小
var Scanner = new Scanner();
var N = Scanner.NextLong();
var W = Scanner.NextLong();
var vec = new List<(int, int)>();

for (var i = 0; i < N; i++)
    vec.Add((Scanner.NextInt(), Scanner.NextInt()));

var dp = new long[N + 1][];
for (var i = 0; i <= N; i++)
    dp[i] = new long[W + 1];

for (var i = 0; i < N; i++)
{
    for (var j = 0; j <= W; j++)
    {
        if (j >= vec[i].Item1)
        {
            dp[i + 1][j] = Max(dp[i][j], dp[i][j - vec[i].Item1] + vec[i].Item2);
        }
        else
        {
            dp[i + 1][j] = dp[i][j];
        }
    }
}
Output(dp[N][W]);
return;



// ナップサックDP  N*W が大
var Scanner = new Scanner();
var N = Scanner.NextLong();
var W = Scanner.NextLong();

var Vmax = 2e5.ToInt();

var vec = new List<(int, int)>();

for (var i = 0; i < N; i++)
    vec.Add((Scanner.NextInt(), Scanner.NextInt()));

var dp = new long[N + 1][];
for (var i = 0; i <= N; i++)
    dp[i] = Enumerable.Repeat(1L << 60, Vmax).ToArray();
dp[0][0] = 0;

for (var i = 1; i <= N; i++)
{
    for (var j = 0; j < Vmax; j++)
    {
        dp[i][j] = dp[i - 1][j];

        if (j >= vec[i - 1].Item2)
        {
            dp[i][j] = Min(dp[i - 1][j], dp[i - 1][j - vec[i - 1].Item2] + vec[i - 1].Item1);
        }
    }
}

var ans = 0L;
for (var i = 0; i < Vmax; i++)
{
    if (dp[N][i] <= W)
        ans = Max(ans, i);
}

Output(ans);
return;
